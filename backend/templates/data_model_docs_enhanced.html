🗄️ SQLite Database Tables (Local Storage)

1. users #

Purpose: Stores user authentication and profile information for system access control.
-- Create users table CREATE TABLE users ( id INTEGER PRIMARY KEY AUTOINCREMENT, username VARCHAR(80) UNIQUE NOT NULL, email VARCHAR(120) UNIQUE NOT NULL, password VARCHAR(200) NOT NULL, role VARCHAR(20) DEFAULT 'user', created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); -- Sample query: Get all admin users SELECT * FROM users WHERE role = 'admin'; -- Sample query: Count users by role SELECT role, COUNT(*) as count FROM users GROUP BY role;
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique user identifier
username VARCHAR(80) UNIQUE NOT NULL User login username
email VARCHAR(120) UNIQUE NOT NULL User email address
password VARCHAR(200) NOT NULL Hashed password (bcrypt)
role VARCHAR(20) DEFAULT: 'user' User role (admin/user/manager)
created_at DATETIME DEFAULT: CURRENT_TIMESTAMP Account creation timestamp

2. orders #

Purpose: Manages customer orders with delivery information and status tracking.
-- Create orders table CREATE TABLE orders ( id INTEGER PRIMARY KEY AUTOINCREMENT, order_date DATE NOT NULL, customer_id INTEGER NOT NULL, product_id TEXT NOT NULL, quantity INTEGER NOT NULL, delivery_address TEXT NOT NULL, latitude REAL NOT NULL, longitude REAL NOT NULL, delivery_time TIMESTAMP, zone TEXT DEFAULT 'A', status VARCHAR(20) DEFAULT 'pending' ); -- Sample query: Get pending orders SELECT * FROM orders WHERE status = 'pending' ORDER BY order_date DESC; -- Sample query: Orders by zone SELECT zone, COUNT(*) as order_count, SUM(quantity) as total_quantity FROM orders GROUP BY zone;
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique order identifier
order_date DATE NOT NULL Order placement date
customer_id INTEGER NOT NULL Customer identifier
product_id TEXT NOT NULL Product SKU or identifier
quantity INTEGER NOT NULL Order quantity
delivery_address TEXT NOT NULL Delivery destination address
latitude REAL NOT NULL Geocoded latitude coordinate
longitude REAL NOT NULL Geocoded longitude coordinate
delivery_time TIMESTAMP Estimated/Actual delivery time
zone TEXT DEFAULT: 'A' Delivery zone classification
status VARCHAR(20) DEFAULT: 'pending' Order status (pending/in_transit/delivered/cancelled)

3. routes #

Purpose: Stores optimized delivery routes with assigned vehicles and performance metrics.
-- Create routes table CREATE TABLE routes ( id INTEGER PRIMARY KEY AUTOINCREMENT, route_name VARCHAR(100), vehicle_id INTEGER, distance FLOAT, estimated_time FLOAT, status VARCHAR(20) DEFAULT 'planned', optimization_score FLOAT, created_at DATETIME DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (vehicle_id) REFERENCES vehicles(id) ); -- Sample query: Active routes with vehicle info SELECT r.*, v.vehicle_number, v.driver_name FROM routes r JOIN vehicles v ON r.vehicle_id = v.id WHERE r.status = 'active';
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique route identifier
route_name VARCHAR(100) Route display name
vehicle_id INTEGER FOREIGN KEY Assigned vehicle reference
distance FLOAT Total route distance in km
estimated_time FLOAT Estimated completion time in hours
status VARCHAR(20) DEFAULT: 'planned' Route status (planned/active/completed)
optimization_score FLOAT AI optimization efficiency score
created_at DATETIME DEFAULT: CURRENT_TIMESTAMP Route creation timestamp

4. vehicles #

Purpose: Manages fleet information including capacity, location, and availability status.
-- Create vehicles table CREATE TABLE vehicles ( id INTEGER PRIMARY KEY AUTOINCREMENT, vehicle_number VARCHAR(50) UNIQUE, vehicle_type VARCHAR(50), capacity FLOAT, current_location TEXT, status VARCHAR(20) DEFAULT 'available', driver_name VARCHAR(100), created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); -- Sample query: Available vehicles by type SELECT vehicle_type, COUNT(*) as available_count FROM vehicles WHERE status = 'available' GROUP BY vehicle_type;
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique vehicle identifier
vehicle_number VARCHAR(50) UNIQUE Vehicle registration/fleet number
vehicle_type VARCHAR(50) Vehicle category (truck/van/bike)
capacity FLOAT Maximum load capacity in kg
current_location TEXT Current vehicle location
status VARCHAR(20) DEFAULT: 'available' Vehicle status (available/in_use/maintenance)
driver_name VARCHAR(100) Assigned driver name
created_at DATETIME DEFAULT: CURRENT_TIMESTAMP Vehicle registration timestamp

5. inventory #

Purpose: Tracks warehouse inventory levels with AI-powered demand forecasting.
-- Create inventory table CREATE TABLE inventory ( id INTEGER PRIMARY KEY AUTOINCREMENT, item_name VARCHAR(100) NOT NULL, sku VARCHAR(50) UNIQUE, quantity INTEGER DEFAULT 0, reorder_level INTEGER, warehouse_location VARCHAR(100), last_restock_date DATETIME, predicted_demand FLOAT, created_at DATETIME DEFAULT CURRENT_TIMESTAMP ); -- Sample query: Items below reorder level SELECT * FROM inventory WHERE quantity <= reorder_level;
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique inventory item identifier
item_name VARCHAR(100) NOT NULL Product/Item name
sku VARCHAR(50) UNIQUE Stock Keeping Unit code
quantity INTEGER DEFAULT: 0 Current stock quantity
reorder_level INTEGER Minimum quantity before reorder
warehouse_location VARCHAR(100) Storage location within warehouse
last_restock_date DATETIME Last inventory replenishment date
predicted_demand FLOAT AI-predicted demand forecast
created_at DATETIME DEFAULT: CURRENT_TIMESTAMP Item creation timestamp

6. demand_history #

Purpose: Historical demand data for machine learning and forecasting models.
-- Create demand_history table CREATE TABLE demand_history ( id INTEGER PRIMARY KEY AUTOINCREMENT, date DATE NOT NULL, zone TEXT NOT NULL, demand INTEGER NOT NULL, weather TEXT, day_of_week INTEGER, is_holiday INTEGER DEFAULT 0 ); -- Sample query: Weekly demand pattern SELECT day_of_week, AVG(demand) as avg_demand FROM demand_history GROUP BY day_of_week ORDER BY day_of_week;
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique record identifier
date DATE NOT NULL Historical data date
zone TEXT NOT NULL Delivery zone
demand INTEGER NOT NULL Number of orders/demand units
weather TEXT Weather conditions
day_of_week INTEGER Day of week (0-6)
is_holiday INTEGER DEFAULT: 0 Holiday flag (0 or 1)

7. model_performance #

Purpose: Tracks AI/ML model training history and performance metrics.
-- Create model_performance table CREATE TABLE model_performance ( id INTEGER PRIMARY KEY AUTOINCREMENT, model_name TEXT NOT NULL, training_date DATETIME DEFAULT CURRENT_TIMESTAMP, accuracy REAL, mae REAL, rmse REAL, parameters TEXT ); -- Sample query: Best performing models SELECT model_name, accuracy, mae, rmse FROM model_performance ORDER BY accuracy DESC LIMIT 5;
Column Name Data Type Constraints Description
id INTEGER PRIMARY KEY Unique record identifier
model_name TEXT NOT NULL ML model identifier
training_date DATETIME DEFAULT: CURRENT_TIMESTAMP Model training timestamp
accuracy REAL Model accuracy score
mae REAL Mean Absolute Error
rmse REAL Root Mean Square Error
parameters TEXT Model hyperparameters (JSON)

🔗 Database Relationships & Data Flow

📌 Key Relationships:
  • orders.customer_id → Customer references
  • routes.vehicle_id → vehicles.id (Many-to-One)
  • demand_history.zone → orders.zone (Lookup relationship)
  • Azure TrackingData.VehicleId → SQLite vehicles.vehicle_number
  • Azure DeliveryLogs.OrderId → SQLite orders.id

Data Flow Architecture

  1. Order Creation: Customer data stored in SQLite orders table
  2. Route Optimization: AI algorithms (Genetic Algorithm/OR-Tools) create optimized routes in routes table
  3. Vehicle Assignment: Available vehicles assigned to routes
  4. Real-time Tracking: GPS data streamed to Azure TrackingData table
  5. Event Logging: All activities recorded in Azure DeliveryLogs
  6. Analytics Processing: AI insights stored in Azure AIAnalytics
  7. Performance Monitoring: KPIs calculated and stored in Azure PerformanceMetrics
  8. Demand Forecasting: Historical data from demand_history used for LSTM predictions

🤖 AI Agents & Learning Systems

Traditional AI Agents

Purpose: Autonomous agents for real-time decision making and optimization.
  • Route Optimization Agent: Uses genetic algorithms and OR-Tools for optimal route planning
  • Demand Forecasting Agent: LSTM-based time series prediction for inventory planning
  • Risk Assessment Agent: RandomForest classifier for delivery risk evaluation
  • Fleet Management Agent: Vehicle allocation and scheduling optimization

GenAI Agents (OpenAI Integration)

Purpose: Natural language processing and intelligent recommendations.
  • Insight Generator: Analyzes data and provides natural language insights
  • Recommendation Engine: Suggests operational improvements based on patterns
  • Query Assistant: Natural language interface for data queries
  • Report Summarizer: Automated report generation and summarization

Self-Learning System (Phase 2 & 3)

Purpose: Continuous learning and model improvement through feedback loops.
  • Adaptive Learning: Models retrain based on new data and performance feedback
  • A/B Testing: Compares algorithm performance and selects best performers
  • Reinforcement Learning: Learns optimal strategies through trial and reward
  • Transfer Learning: Applies knowledge from one domain to improve others
